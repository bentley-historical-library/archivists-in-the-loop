{{- /* Returns a slice of schema.org Person dicts from a report's `authors` or `editors` front matter. Each entry may be a map ({name, orcid}) or a plain string. Entries with no name are skipped. An orcid (numbers only) becomes the Person's identifier URL. */ -}} {{- $people := slice -}} {{- range . -}} {{- $name := "" -}} {{- $orcid := "" -}} {{- if reflect.IsMap . }}{{ $name = .name }}{{ $orcid = .orcid }}{{ else }}{{ $name = . }}{{ end -}} {{- if $name -}} {{- $person := dict "@type" "Person" "name" $name -}} {{- with $orcid }}{{ $person = merge $person (dict "identifier" (printf "https://orcid.org/%s" (strings.TrimPrefix "https://orcid.org/" .))) }}{{ end -}} {{- $people = $people | append $person -}} {{- end -}} {{- end -}} {{- return $people -}}
